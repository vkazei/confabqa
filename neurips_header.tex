% NeurIPS-look styling injected into pandoc's default LaTeX template.
% Reproduces the visible style of neurips_2024.sty: 10pt Times body,
% Helvetica sans headings, 5.5in x 9in text area, NeurIPS-style abstract.

% Fonts: use fontspec under xelatex with system fonts that have Cyrillic + CJK
% coverage. NeurIPS uses Times Roman body + Helvetica headings.
\usepackage{mathptmx}                 % math + body Times
\usepackage[scaled=0.92]{helvet}      % sans default = Helvetica scaled
% Override main font for text to a Cyrillic-capable Times variant
\usepackage{fontspec}
\setmainfont{Times New Roman}
\setsansfont[Scale=0.92]{Helvetica Neue}
\setmonofont[Scale=0.85, HyphenChar=None]{Menlo}

% Section formatting: sans-serif bold, tight NeurIPS spacing.
\usepackage[compact]{titlesec}
\titleformat*{\section}{\large\bfseries\sffamily}
\titleformat*{\subsection}{\normalsize\bfseries\sffamily}
\titleformat*{\subsubsection}{\normalsize\bfseries\sffamily}
\titleformat*{\paragraph}{\normalsize\bfseries\sffamily}
\titlespacing*{\section}{0pt}{1.6ex plus 0.3ex minus 0.2ex}{0.6ex plus 0.1ex}
\titlespacing*{\subsection}{0pt}{1.2ex plus 0.3ex minus 0.2ex}{0.4ex plus 0.1ex}
\titlespacing*{\subsubsection}{0pt}{1.0ex plus 0.2ex minus 0.2ex}{0.3ex plus 0.1ex}

% Paragraph spacing: zero parskip, tighter line spacing.
\setlength{\parindent}{1em}
\setlength{\parskip}{0pt}
\linespread{0.97}

% Tighter list spacing.
\usepackage{enumitem}
\setlist{topsep=2pt, partopsep=0pt, itemsep=1pt, parsep=0pt}

% NeurIPS-style abstract: tighter spacing, narrower indent.
\renewenvironment{abstract}{%
  \vspace{0.6ex}%
  \noindent\rule{\textwidth}{0.4pt}\par
  \vspace{0.2ex}%
  \begin{center}{\normalsize\bfseries\sffamily Abstract}\end{center}%
  \vspace{-0.2ex}%
  \noindent\rule{\textwidth}{0.4pt}\par
  \vspace{0.4ex}%
  \begingroup
    \leftskip=1em \rightskip=1em
    \small
    \noindent\ignorespaces
}{%
  \par\endgroup
  \vspace{0.3ex}%
  \noindent\rule{\textwidth}{0.4pt}\par
  \vspace{1ex}%
}

% Sans-serif title (NeurIPS look)
\usepackage{titling}
\pretitle{\begin{center}\LARGE\bfseries\sffamily}
\posttitle{\par\end{center}\vspace{0.5em}}
\preauthor{\begin{center}\normalsize\sffamily}
\postauthor{\par\end{center}\vspace{0.3em}}
\predate{\begin{center}\normalsize\itshape}
\postdate{\par\end{center}\vspace{1em}}

% Compact figure / table captions.
\usepackage{caption}
\captionsetup{font=footnotesize,labelfont=bf,skip=3pt}

% Tighter table row spacing + smaller table font.
\renewcommand{\arraystretch}{1.0}
\AtBeginEnvironment{tabular}{\small}
\AtBeginEnvironment{longtable}{\small}

% Float spacing.
\setlength{\textfloatsep}{6pt plus 2pt minus 2pt}
\setlength{\intextsep}{6pt plus 2pt minus 2pt}
\setlength{\floatsep}{6pt plus 2pt minus 2pt}

% Reduce orphans/widows
\widowpenalty=10000
\clubpenalty=10000

% TikZ for vector schematics (Fig. 1 methods diagram).
\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, calc, shapes.geometric}

% xeCJK (loaded by the template when CJKmainfont is set) claims curly quotes and
% dashes as CJK punctuation, so they get typeset in Heiti SC with fullwidth-ish
% advance widths ("Llama' s"). Reclassify them as Latin so Times New Roman keeps them.
\AtBeginDocument{%
  \ifcsname xeCJKDeclareCharClass\endcsname
    \xeCJKDeclareCharClass{Default}{"2018, "2019, "201C, "201D, "2013, "2014}%
  \fi
}

% Cyrillic passthrough (arXiv header maps this to a fallback font).
\newcommand{\cyrtext}[1]{#1}

% Code-heavy paragraphs: with unbreakable \texttt tokens, the default
% \tolerance=200 makes TeX run tokens into the margin rather than set a
% loose line. Prefer the loose line.
\tolerance=2400
\hbadness=2400
